cmake_minimum_required(VERSION 3.24)
project(CyCodeReader VERSION 0.1.7 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" /utf-8)
  # Keep source diagnostics useful without embedding the developer checkout path.
  add_compile_options("$<$<CONFIG:Release>:/experimental:deterministic>" "$<$<CONFIG:Release>:/pathmap:${CMAKE_CURRENT_SOURCE_DIR}=.>")
endif()
option(BUILD_TESTING "Build opt-in Reader native test executables" OFF)
include(CTest)
option(CYCODE_READER_BUILD_GUI "Build the native Reader application" ON)
option(CYCODE_READER_PUBLIC_UPDATES "Enable unauthenticated public GitHub release checks" OFF)
set(CYCODE_READER_UPDATE_REPOSITORY "MrMybal/CyCodeReader" CACHE STRING "Public GitHub owner/repository")
if(NOT CYCODE_READER_UPDATE_REPOSITORY MATCHES "^[A-Za-z0-9_-]+/[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "Update repository must be a GitHub owner/repository")
endif()
configure_file(packaging/distribution-build.json.in "${CMAKE_BINARY_DIR}/distribution-build.json" @ONLY)
set(CYCODE_EDITOR_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/deps" CACHE PATH "Pinned build dependencies")
set(CYUI_DIR "" CACHE PATH "Optional clean CyUI checkout at the locked revision (empty: fetch pinned source)")
if(MSVC)
  add_compile_options("$<$<CONFIG:Release>:/pathmap:${CYCODE_EDITOR_DEPS}=external>")
endif()
set(CYCODE_EDITOR_BUILD_GUI ${CYCODE_READER_BUILD_GUI})
set(CYCODE_EDITOR_UI_MODULES "CodeEditor;Commands;Inspector;Diagnostics;Diff")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CYCODE_CORE_ROOT "" CACHE PATH "Explicit native CyCode source checkout")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/editor/scripts/core_dependency.py"
  --source "${CYCODE_CORE_ROOT}" OUTPUT_VARIABLE CYCODE_CORE_ROOT
  OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT EXISTS "${CYCODE_CORE_ROOT}/cmake/CyCodeComponents.cmake")
  message(FATAL_ERROR "Reader requires the modular native CyCode source API; use core-source.lock.json")
endif()
set(CYCODE_COMMON_DEPS "${CYCODE_EDITOR_DEPS}")
include("${CYCODE_CORE_ROOT}/cmake/CyCodeComponents.cmake")
cycode_add_components(Documents TextSession TextBridge Process Snippets PagedText)
add_subdirectory(editor)
add_library(cycode_reader_documents INTERFACE)
target_include_directories(cycode_reader_documents INTERFACE src)
target_link_libraries(cycode_reader_documents INTERFACE CyCode::Documents)
find_package(Threads REQUIRED)
add_library(cycode_reader_distribution STATIC src/Distribution.cpp src/Updates.cpp src/PortableUpdate.cpp)
target_link_libraries(cycode_reader_distribution PUBLIC cycode_files Threads::Threads)
target_include_directories(cycode_reader_distribution PUBLIC src)
target_compile_definitions(cycode_reader_distribution PUBLIC
  CYCODE_READER_VERSION="${PROJECT_VERSION}"
  CYCODE_READER_UPDATE_REPOSITORY="${CYCODE_READER_UPDATE_REPOSITORY}"
  CYCODE_READER_PUBLIC_UPDATES=$<BOOL:${CYCODE_READER_PUBLIC_UPDATES}>)
if(WIN32)
  target_compile_definitions(cycode_reader_distribution PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX)
  target_link_libraries(cycode_reader_distribution PUBLIC winhttp bcrypt shell32 ole32)
  add_executable(CyCodeReaderUpdater WIN32 src/updater_main.cpp)
  target_link_libraries(CyCodeReaderUpdater PRIVATE cycode_reader_distribution CyCode::Process)
endif()
add_library(cycode_reader_core INTERFACE)
target_link_libraries(cycode_reader_core INTERFACE cycode_reader_documents CyCode::TextSession CyCode::TextBridge CyCode::Process)
add_executable(cycodereader_cli src/cli.cpp)
target_link_libraries(cycodereader_cli PRIVATE cycode_reader_core)
if(WIN32)
  target_link_libraries(cycodereader_cli PRIVATE shell32)
endif()
add_library(reader_sample_plugin MODULE examples/sample_plugin.cpp)
target_include_directories(reader_sample_plugin PRIVATE src)
target_link_libraries(reader_sample_plugin PRIVATE CyCode::TextContracts)
add_library(reader_git_plugin MODULE plugins/git/GitPlugin.cpp)
target_include_directories(reader_git_plugin PRIVATE src)
target_link_libraries(reader_git_plugin PRIVATE cycode_reader_documents CyCode::Process Threads::Threads)
set_target_properties(reader_git_plugin PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
set_target_properties(reader_sample_plugin PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${configuration}" upper)
  set_target_properties(reader_sample_plugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/plugins" RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/plugins")
  set_target_properties(reader_git_plugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/plugins" RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/plugins")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
configure_file(plugins/text-tools.json "${CMAKE_BINARY_DIR}/plugins/text-tools.json" COPYONLY)
configure_file(plugins/native-sample.json "${CMAKE_BINARY_DIR}/plugins/native-sample.json" COPYONLY)
configure_file(plugins/git.json "${CMAKE_BINARY_DIR}/plugins/git.json" COPYONLY)

    set_tests_properties(reader_git_plugin PROPERTIES TIMEOUT 90)
  endif()
  add_test(NAME reader_distribution_fixtures COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/distribution_fixtures.py" "${CMAKE_BINARY_DIR}/distribution-fixtures")

  set_tests_properties(reader_distribution PROPERTIES TIMEOUT 30 FIXTURES_REQUIRED reader_distribution_packages)
  add_executable(cycode_reader_document_tests tests/Documents.cpp)
  target_link_libraries(cycode_reader_document_tests PRIVATE cycode_reader_documents)
  add_test(NAME reader_documents COMMAND cycode_reader_document_tests)

  add_executable(cycode_reader_core_tests tests/Core.cpp)
  target_link_libraries(cycode_reader_core_tests PRIVATE cycode_reader_core)

  add_test(NAME reader_native_bridge COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_bridge.py" "$<TARGET_FILE:cycodereader_cli>")

  set_tests_properties(reader_core PROPERTIES TIMEOUT 20)

if(CYCODE_READER_BUILD_GUI)
  add_library(cycode_reader_editor STATIC src/LiveEditor.cpp src/PluginLanguage.cpp src/ModuleModels.cpp)
  target_link_libraries(cycode_reader_editor PUBLIC cycode_reader_core cycode_editor)
  target_link_libraries(cycode_reader_editor PUBLIC CyUI::Commands CyUI::Inspector CyUI::Diagnostics)
  add_library(cycode_reader_ui STATIC src/Reader.cpp src/PagedUI.cpp src/ServicesUI.cpp src/Menus.cpp src/UpdatesUI.cpp src/Branding.cpp src/ComparisonUI.cpp src/SearchUI.cpp src/HistoryUI.cpp src/DiffView.cpp)
  target_link_libraries(cycode_reader_ui PUBLIC cycode_reader_editor cycode_fileicons cycode_reader_distribution CyUI::Diff)
  if(BUILD_TESTING)
    add_executable(cycode_reader_experience_tests tests/Experience.cpp)
    target_link_libraries(cycode_reader_experience_tests PRIVATE cycode_reader_ui)
    add_test(NAME reader_experience COMMAND cycode_reader_experience_tests "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(reader_experience PROPERTIES TIMEOUT 45)
    if(GIT_FOUND)
      add_executable(cycode_reader_comparison_tests tests/Comparison.cpp)
      target_link_libraries(cycode_reader_comparison_tests PRIVATE cycode_reader_ui)
      add_dependencies(cycode_reader_comparison_tests reader_git_plugin)

    endif()
    add_executable(cycode_reader_update_ui_tests tests/UpdatesUI.cpp)
    target_link_libraries(cycode_reader_update_ui_tests PRIVATE cycode_reader_ui)
    add_test(NAME reader_updates_ui COMMAND cycode_reader_update_ui_tests "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(reader_updates_ui PROPERTIES TIMEOUT 30)
    add_executable(cycode_reader_editor_tests tests/EditorIntegration.cpp)
    target_link_libraries(cycode_reader_editor_tests PRIVATE cycode_reader_editor)

    set_tests_properties(reader_editor_integration PROPERTIES TIMEOUT 20)

    target_link_libraries(cycode_reader_module_tests PRIVATE cycode_reader_ui)
    add_test(NAME reader_cyui_modules COMMAND cycode_reader_module_tests "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(reader_cyui_modules PROPERTIES TIMEOUT 45)

    target_link_libraries(cycode_reader_filter_tests PRIVATE cycode_reader_ui)

    set_tests_properties(reader_filtering PROPERTIES TIMEOUT 45)

  endif()
  add_executable(CyCodeReader src/main.cpp)
  set_property(TARGET CyCodeReader APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/fonts/JetBrainsMono-Regular.ttf"
    "${CMAKE_CURRENT_SOURCE_DIR}/licenses/JetBrainsMono-OFL.txt"
    "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader.ico"
    "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader-document.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader-document.ico")
  target_link_libraries(CyCodeReader PRIVATE cycode_reader_ui)
  option(CYCODE_READER_BUILD_DOCUMENTATION "Generate and distribute the local CyDoc manual" ON)
  if(CYCODE_READER_BUILD_DOCUMENTATION)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_custom_target(CyCodeReader_documentation
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/.CyDocBuild/build.py"
        --output "${CMAKE_BINARY_DIR}/Documentation" --output "$<TARGET_FILE_DIR:CyCodeReader>/Documentation"
      WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
      COMMENT "Compiling the offline CyDoc manual" VERBATIM)
    add_dependencies(cycode_reader_ui CyCodeReader_documentation)
    target_include_directories(cycode_reader_ui PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.CyDocBuild/node_modules/cydoc/examples/integrations")
    target_compile_definitions(cycode_reader_ui PRIVATE CYCODE_LOCAL_HELP)
    if(WIN32)
      target_link_libraries(cycode_reader_ui PRIVATE shlwapi)
    endif()
    install(DIRECTORY "${CMAKE_BINARY_DIR}/Documentation/" DESTINATION Documentation)

  add_dependencies(CyCodeReader reader_sample_plugin reader_git_plugin)
  if(WIN32)
    enable_language(RC)
    configure_file(packaging/version.rc.in "${CMAKE_BINARY_DIR}/reader-version.rc" @ONLY)
    target_sources(CyCodeReader PRIVATE "${CMAKE_BINARY_DIR}/reader-version.rc")
    target_sources(CyCodeReaderUpdater PRIVATE "${CMAKE_BINARY_DIR}/reader-version.rc")
    add_dependencies(CyCodeReader CyCodeReaderUpdater)
    set_target_properties(CyCodeReader PROPERTIES WIN32_EXECUTABLE TRUE)
    if(MSVC)
      target_link_options(CyCodeReader PRIVATE /ENTRY:mainCRTStartup)
    endif()
    target_link_libraries(CyCodeReader PRIVATE shell32)
  endif()
  add_custom_command(TARGET CyCodeReader POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:CyCodeReader>/locales"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/locales/ui.json" "$<TARGET_FILE_DIR:CyCodeReader>/locales/ui.json"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:CyCodeReader>/app"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/app/tool.py" "${CMAKE_CURRENT_SOURCE_DIR}/app/contracts.py" "${CMAKE_CURRENT_SOURCE_DIR}/app/CyTool.json" "$<TARGET_FILE_DIR:CyCodeReader>/app"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_BINARY_DIR}/plugins" "$<TARGET_FILE_DIR:CyCodeReader>/plugins"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:CyCodeReader>/licenses"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:CyCodeReader>/fonts"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/fonts/JetBrainsMono-Regular.ttf" "$<TARGET_FILE_DIR:CyCodeReader>/fonts"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:CyCodeReader>/icons"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader.png" "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader.ico" "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader-document.png" "${CMAKE_CURRENT_SOURCE_DIR}/icons/CyCodeReader-document.ico" "$<TARGET_FILE_DIR:CyCodeReader>/icons"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/COPYRIGHT" "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "$<TARGET_FILE_DIR:CyCodeReader>")
  install(TARGETS CyCodeReader RUNTIME DESTINATION .)
  if(WIN32)
    install(TARGETS CyCodeReaderUpdater RUNTIME DESTINATION .)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
    include(InstallRequiredSystemLibraries)
  endif()
  install(FILES LICENSE COPYRIGHT THIRD_PARTY_NOTICES.md DESTINATION .)
  install(FILES locales/ui.json DESTINATION locales)
  install(DIRECTORY app/ DESTINATION app FILES_MATCHING PATTERN "*.py" PATTERN "*.json")
  install(DIRECTORY "${CMAKE_BINARY_DIR}/plugins/" DESTINATION plugins FILES_MATCHING PATTERN "*.json" PATTERN "*.dll" PATTERN "*.so" PATTERN "*.dylib")
  install(FILES fonts/JetBrainsMono-Regular.ttf DESTINATION fonts)
  install(DIRECTORY licenses/ DESTINATION licenses)
  install(FILES icons/CyCodeReader.png icons/CyCodeReader.ico icons/CyCodeReader-document.png icons/CyCodeReader-document.ico DESTINATION icons)
endif()
