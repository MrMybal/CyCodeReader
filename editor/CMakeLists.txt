# Source library shared by CyCodeReader and CyCode. No IDE services are linked.
set(CYCODE_EDITOR_API_VERSION 1)
if(NOT DEFINED CYCODE_EDITOR_DEPS)
  get_filename_component(CYCODE_EDITOR_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/deps" ABSOLUTE)
endif()
option(CYCODE_EDITOR_BUILD_GUI "Build the shared graphical editor" ON)
# The IDE may already supply Core. Resolve once and validate its source identity.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(TARGET CyCode::Core)
  # The embedding IDE already selected its native source revision. Its freshly
  # compiled Core and this editor must share that one source graph; the Reader
  # standalone lock is not a second selection for the consuming application.
  get_filename_component(CYCODE_CORE_ROOT "${CYCODE_CORE_ROOT}" REALPATH)
  get_target_property(_cycode_host_source cycode_core SOURCE_DIR)
  get_filename_component(_cycode_host_source "${_cycode_host_source}/.." REALPATH)
  if(NOT _cycode_host_source STREQUAL CYCODE_CORE_ROOT)
    message(FATAL_ERROR "The host Core target and supplied native source differ")
  endif()
  foreach(_cycode_api api-version.txt common/api-version.txt)
    file(READ "${CYCODE_CORE_ROOT}/${_cycode_api}" _cycode_version)
    string(STRIP "${_cycode_version}" _cycode_version)
    if(NOT _cycode_version STREQUAL "1")
      message(FATAL_ERROR "The shared editor requires native/common source API 1")
    endif()
  endforeach()
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYCODE_CORE_ROOT}"
    -C "${CYCODE_CORE_ROOT}" status --porcelain --untracked-files=no
    OUTPUT_VARIABLE _cycode_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(_cycode_dirty)
    message(FATAL_ERROR "The embedding host Core has tracked local changes; preserve them and select a clean source")
  endif()
else()
  # Standalone Reader and editor-only builds retain the exact approved lock.
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/core_dependency.py"
    --source "${CYCODE_CORE_ROOT}"
    OUTPUT_VARIABLE CYCODE_CORE_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
set(CYCODE_COMMON_DEPS "${CYCODE_EDITOR_DEPS}")
include("${CYCODE_CORE_ROOT}/cmake/CyCodeComponents.cmake")
cycode_add_components(Files Snippets Documents PagedText)

if(BUILD_TESTING)
  add_executable(cycode_snippet_tests tests/snippet_tests.cpp)
  target_link_libraries(cycode_snippet_tests PRIVATE cycode_snippets)
  add_test(NAME snippet_expansion COMMAND cycode_snippet_tests)
endif()
if(CYCODE_EDITOR_BUILD_GUI)
  find_package(Threads REQUIRED)
  find_package(OpenGL REQUIRED)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYCODE_EDITOR_DEPS}/SDL" "${CMAKE_CURRENT_BINARY_DIR}/SDL" EXCLUDE_FROM_ALL)
  set(IMGUI_DIR "${CYCODE_EDITOR_DEPS}/imgui")
  add_library(cycode_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp" "${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp")
  target_include_directories(cycode_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends" "${IMGUI_DIR}/misc/cpp")
  target_compile_definitions(cycode_imgui PUBLIC IMGUI_USE_WCHAR32)
  target_link_libraries(cycode_imgui PUBLIC SDL3::SDL3-static OpenGL::GL)
  set(CYUI_IMGUI_TARGET cycode_imgui CACHE STRING "" FORCE)
  set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_WEB_CEF OFF CACHE BOOL "" FORCE)
  if(NOT DEFINED CYCODE_EDITOR_UI_MODULES)
    set(CYCODE_EDITOR_UI_MODULES "CodeEditor")
  endif()
  set(CYUI_MODULES "${CYCODE_EDITOR_UI_MODULES}" CACHE STRING "" FORCE)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  option(CYCODE_CYUI_DEVELOPMENT "Use an explicit local development CyUI checkout without changing its source lock" OFF)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cyui-source.lock.json" CYCODE_CYUI_LOCK)
  string(JSON CYCODE_CYUI_REVISION GET "${CYCODE_CYUI_LOCK}" revision)
  string(SUBSTRING "${CYCODE_CYUI_REVISION}" 0 12 CYCODE_CYUI_CACHE_ID)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/cyui_dependency.py"
    --destination "${CYCODE_EDITOR_DEPS}/cyui-${CYCODE_CYUI_CACHE_ID}" --source "${CYUI_DIR}" --allow-development "${CYCODE_CYUI_DEVELOPMENT}"
    OUTPUT_VARIABLE CYUI_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  add_subdirectory("${CYUI_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cyui")
  # CyUI currently has no host-backend option. Keep our existing native hooks in a
  # build-local copy of its MIT backend, shared by CyUI and every host consumer.
  # No CyUI source or downloaded source is modified; fail on changed patch anchors.
  set(CYCODE_EDITOR_DIR "${CMAKE_CURRENT_BINARY_DIR}/texteditor")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/cyui-source.lock.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    "${CYUI_DIR}/third_party/texteditor/TextEditor.h" "${CYUI_DIR}/third_party/texteditor/TextEditor.cpp")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    --source "${CYUI_DIR}/third_party/texteditor" --destination "${CYCODE_EDITOR_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  set_property(TARGET cyui_texteditor PROPERTY SOURCES "${CYCODE_EDITOR_DIR}/TextEditor.cpp")
  target_include_directories(cyui_texteditor BEFORE PUBLIC "${CYCODE_EDITOR_DIR}")
  add_library(cycode_texteditor ALIAS cyui_texteditor)
  add_library(cycode_tree_sitter STATIC "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/src/lib.c")
  target_include_directories(cycode_tree_sitter PUBLIC "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/include" PRIVATE "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/src")
  set_target_properties(cycode_tree_sitter PROPERTIES C_STANDARD 11)
  foreach(grammar rust java javascript html css)
    set(grammar_root "${CYCODE_EDITOR_DEPS}/tree-sitter-${grammar}/src")
    add_library(cycode_grammar_${grammar} STATIC "${grammar_root}/parser.c")
    if(EXISTS "${grammar_root}/scanner.c")
      target_sources(cycode_grammar_${grammar} PRIVATE "${grammar_root}/scanner.c")
    endif()
    target_include_directories(cycode_grammar_${grammar} PRIVATE "${grammar_root}")
    target_link_libraries(cycode_tree_sitter PUBLIC cycode_grammar_${grammar})
  endforeach()
  foreach(grammar typescript tsx)
    set(grammar_root "${CYCODE_EDITOR_DEPS}/tree-sitter-typescript/${grammar}/src")
    add_library(cycode_grammar_${grammar} STATIC "${grammar_root}/parser.c" "${grammar_root}/scanner.c")
    target_include_directories(cycode_grammar_${grammar} PRIVATE "${grammar_root}")
    target_link_libraries(cycode_tree_sitter PUBLIC cycode_grammar_${grammar})
  endforeach()
  add_library(cycode_editor STATIC src/PagedTextEditor.cpp src/EditorTheme.cpp src/EditorLanguage.cpp src/EditorHighlight.cpp src/EditorCompletion.cpp src/EditorSnippet.cpp src/Syntax.cpp)
  target_include_directories(cycode_editor PUBLIC include "${CYCODE_EDITOR_DEPS}/json/single_include")
  target_link_libraries(cycode_editor PUBLIC CyUI::Core CyUI::Code CyUI::CodeEditor CyCode::PagedText CyCode::Documents Threads::Threads cycode_tree_sitter cycode_snippets)
  if(BUILD_TESTING)
    # Native opt-in fixture: pass a new synthetic directory when execution is authorized.
    add_executable(cycode_paged_editor_tests tests/paged_text_editor_tests.cpp)
    target_link_libraries(cycode_paged_editor_tests PRIVATE cycode_editor)
    add_executable(cycode_editor_snippet_tests tests/editor_snippet_tests.cpp)
    target_link_libraries(cycode_editor_snippet_tests PRIVATE cycode_editor)
    add_test(NAME editor_snippets COMMAND cycode_editor_snippet_tests)
    set_tests_properties(editor_snippets PROPERTIES TIMEOUT 30)
    add_executable(cycode_editor_completion_tests tests/editor_completion_tests.cpp)
    target_link_libraries(cycode_editor_completion_tests PRIVATE cycode_editor)
    add_test(NAME editor_completion COMMAND cycode_editor_completion_tests)
    set_tests_properties(editor_completion PROPERTIES TIMEOUT 30)
    add_executable(cycode_syntax_tests tests/syntax_tests.cpp)
    target_link_libraries(cycode_syntax_tests PRIVATE cycode_editor)
    add_test(NAME multilingual_syntax COMMAND cycode_syntax_tests)
    set_tests_properties(multilingual_syntax PROPERTIES TIMEOUT 30)
    add_executable(cycode_highlight_tests tests/highlighting.cpp)
    target_link_libraries(cycode_highlight_tests PRIVATE cycode_editor)
    add_test(NAME editor_highlighting COMMAND cycode_highlight_tests)
    set_tests_properties(editor_highlighting PROPERTIES TIMEOUT 30)
  endif()
  add_library(cycode_fileicons STATIC src/FileIcons.cpp)
  target_link_libraries(cycode_fileicons PUBLIC cycode_editor cycode_files)
endif()
