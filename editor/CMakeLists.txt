# Source library shared by CyCodeReader and CyCode. No IDE services are linked.
set(CYCODE_EDITOR_API_VERSION 1)
if(NOT DEFINED CYCODE_EDITOR_DEPS)
  get_filename_component(CYCODE_EDITOR_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/deps" ABSOLUTE)
endif()
option(CYCODE_EDITOR_BUILD_GUI "Build the shared graphical editor" ON)
add_library(cycode_files STATIC src/Paths.cpp src/TextFile.cpp)
target_include_directories(cycode_files PUBLIC include "${CYCODE_EDITOR_DEPS}/json/single_include")
target_compile_features(cycode_files PUBLIC cxx_std_20)
if(WIN32)
  target_compile_definitions(cycode_files PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(cycode_files PUBLIC /utf-8 /permissive- /Zc:__cplusplus)
endif()
add_library(cycode_snippets STATIC src/Snippet.cpp)
target_include_directories(cycode_snippets PUBLIC include "${CYCODE_EDITOR_DEPS}/json/single_include")
target_link_libraries(cycode_snippets PUBLIC cycode_files)
if(BUILD_TESTING)
  add_executable(cycode_snippet_tests tests/snippet_tests.cpp)
  target_link_libraries(cycode_snippet_tests PRIVATE cycode_snippets)
  add_test(NAME snippet_expansion COMMAND cycode_snippet_tests)
endif()
if(CYCODE_EDITOR_BUILD_GUI)
  find_package(OpenGL REQUIRED)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYCODE_EDITOR_DEPS}/SDL" "${CMAKE_CURRENT_BINARY_DIR}/SDL" EXCLUDE_FROM_ALL)
  set(IMGUI_DIR "${CYCODE_EDITOR_DEPS}/imgui")
  add_library(cycode_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/imgui_demo.cpp" "${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp" "${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp")
  target_include_directories(cycode_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends" "${IMGUI_DIR}/misc/cpp")
  target_compile_definitions(cycode_imgui PUBLIC IMGUI_USE_WCHAR32)
  target_link_libraries(cycode_imgui PUBLIC SDL3::SDL3-static OpenGL::GL)
  set(CYUI_IMGUI_TARGET cycode_imgui CACHE STRING "" FORCE)
  set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYUI_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cyui")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(CYCODE_EDITOR_DIR "${CMAKE_CURRENT_BINARY_DIR}/texteditor")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    "${CYCODE_EDITOR_DEPS}/texteditor/TextEditor.h" "${CYCODE_EDITOR_DEPS}/texteditor/TextEditor.cpp")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_texteditor.py"
    --source "${CYCODE_EDITOR_DEPS}/texteditor" --destination "${CYCODE_EDITOR_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  add_library(cycode_texteditor STATIC "${CYCODE_EDITOR_DIR}/TextEditor.cpp")
  target_include_directories(cycode_texteditor PUBLIC "${CYCODE_EDITOR_DIR}")
  target_link_libraries(cycode_texteditor PUBLIC cycode_imgui)
  add_library(cycode_tree_sitter STATIC "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/src/lib.c")
  target_include_directories(cycode_tree_sitter PUBLIC "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/include" PRIVATE "${CYCODE_EDITOR_DEPS}/tree-sitter/lib/src")
  set_target_properties(cycode_tree_sitter PROPERTIES C_STANDARD 11)
  foreach(grammar rust java javascript html css)
    set(grammar_root "${CYCODE_EDITOR_DEPS}/tree-sitter-${grammar}/src")
    add_library(cycode_grammar_${grammar} STATIC "${grammar_root}/parser.c")
    if(EXISTS "${grammar_root}/scanner.c")
      target_sources(cycode_grammar_${grammar} PRIVATE "${grammar_root}/scanner.c")
    endif()
    target_include_directories(cycode_grammar_${grammar} PRIVATE "${grammar_root}")
    target_link_libraries(cycode_tree_sitter PUBLIC cycode_grammar_${grammar})
  endforeach()
  foreach(grammar typescript tsx)
    set(grammar_root "${CYCODE_EDITOR_DEPS}/tree-sitter-typescript/${grammar}/src")
    add_library(cycode_grammar_${grammar} STATIC "${grammar_root}/parser.c" "${grammar_root}/scanner.c")
    target_include_directories(cycode_grammar_${grammar} PRIVATE "${grammar_root}")
    target_link_libraries(cycode_tree_sitter PUBLIC cycode_grammar_${grammar})
  endforeach()
  add_library(cycode_editor STATIC src/EditorTheme.cpp src/EditorLanguage.cpp src/EditorHighlight.cpp src/EditorCompletion.cpp src/EditorSnippet.cpp src/Syntax.cpp)
  target_include_directories(cycode_editor PUBLIC include "${CYCODE_EDITOR_DEPS}/json/single_include")
  target_link_libraries(cycode_editor PUBLIC CyUI::CyUI cycode_texteditor cycode_tree_sitter cycode_snippets)
  if(BUILD_TESTING)
    add_executable(cycode_editor_snippet_tests tests/editor_snippet_tests.cpp)
    target_link_libraries(cycode_editor_snippet_tests PRIVATE cycode_editor)
    add_test(NAME editor_snippets COMMAND cycode_editor_snippet_tests)
    set_tests_properties(editor_snippets PROPERTIES TIMEOUT 30)
    add_executable(cycode_editor_completion_tests tests/editor_completion_tests.cpp)
    target_link_libraries(cycode_editor_completion_tests PRIVATE cycode_editor)
    add_test(NAME editor_completion COMMAND cycode_editor_completion_tests)
    set_tests_properties(editor_completion PROPERTIES TIMEOUT 30)
    add_executable(cycode_syntax_tests tests/syntax_tests.cpp)
    target_link_libraries(cycode_syntax_tests PRIVATE cycode_editor)
    add_test(NAME multilingual_syntax COMMAND cycode_syntax_tests)
    set_tests_properties(multilingual_syntax PROPERTIES TIMEOUT 30)
    add_executable(cycode_highlight_tests tests/highlighting.cpp)
    target_link_libraries(cycode_highlight_tests PRIVATE cycode_editor)
    add_test(NAME editor_highlighting COMMAND cycode_highlight_tests)
    set_tests_properties(editor_highlighting PROPERTIES TIMEOUT 30)
  endif()
  add_library(cycode_fileicons STATIC src/FileIcons.cpp)
  target_link_libraries(cycode_fileicons PUBLIC cycode_editor cycode_files)
endif()
